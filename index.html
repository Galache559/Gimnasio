function startTimer(seconds) {

updateTimerDisplay(timeLeft);

updateTimerDisplay(timeLeft);

clearInterval(timerInterval);

alert('¡Tiempo de descanso cumplido!');

}

}, 1000);

}


function stopTimer() {

clearInterval(timerInterval);

updateTimerDisplay(0);

}


function updateTimerDisplay(seconds) {

const mins = Math.floor(seconds / 60).toString().padStart(2, '0');

const secs = (seconds % 60).toString().padStart(2, '0');

document.getElementById('timer-display').textContent = `${mins}:${secs}`;

}


function updateMetrics() {

document.getElementById('volumen-total').textContent = `${totalVol.toLocaleString()} kg`;

document.getElementById('total-series').textContent = seriesHoy.length;

}


function calcularResumenesAnalytics() {

const ahora = new Date();

const hace15Dias = new Date();

hace15Dias.setDate(ahora.getDate() - 15);


const hace30Dias = new Date();

hace30Dias.setDate(ahora.getDate() - 30);


// Quincenal (15 Días)

const series15Dias = series.filter(s => new Date(s.fecha) >= hace15Dias);

const vol15 = series15Dias.reduce((acc, s) => acc + (s.peso * s.reps), 0);

document.getElementById('quin-volumen').textContent = `${vol15.toLocaleString()} kg`;

document.getElementById('quin-series').textContent = series15Dias.length;


// Mensual (30 Días)

const series30Dias = series.filter(s => new Date(s.fecha) >= hace30Dias);

const vol30 = series30Dias.reduce((acc, s) => acc + (s.peso * s.reps), 0);

document.getElementById('mes-volumen').textContent = `${vol30.toLocaleString()} kg`;

document.getElementById('mes-series').textContent = series30Dias.length;


// Desglose por ejercicio del último mes

const desgloseEl = document.getElementById('desglose-ejercicios-mes');

desgloseEl.innerHTML = '';


if (series30Dias.length === 0) {

desgloseEl.innerHTML = '<p class="text-xs text-slate-500 py-2">Sin entrenamientos registrados en los últimos 30 días.</p>';

return;

}


const ejercicioMap = {};

series30Dias.forEach(s => {

const vol = s.peso * s.reps;

ejercicioMap[s.ejercicio] = (ejercicioMap[s.ejercicio] || 0) + vol;

});


const ordenados = Object.entries(ejercicioMap).sort((a, b) => b[1] - a[1]);


ordenados.forEach(([nombre, vol]) => {

const item = document.createElement('div');

item.className = 'flex justify-between items-center bg-slate-900/60 p-2 rounded-lg border border-slate-700/60 text-xs';

exportBtn.addEventListener('click', () => {

if (series.length === 0) {

alert('No hay registros guardados para exportar.');

return;

}


let csvContent = "data:text/csv;charset=utf-8,Fecha,Hora,Ejercicio,Peso (kg),Repeticiones,Volumen (kg)\n";

series.forEach(s => {

const vol = s.peso * s.reps;

csvContent += `"${s.fecha || ''}","${s.hora}","${s.ejercicio}",${s.peso},${s.reps},${vol}\n`;

});


const encodedUri = encodeURI(csvContent);

const link = document.createElement("a");

link.setAttribute("href", encodedUri);

const filename = `gym_tracker_4dias_${new Date().toISOString().split('T')[0]}.csv`;

link.setAttribute("download", filename);

document.body.appendChild(link);

link.click();

document.body.removeChild(link);

});


clearBtn.addEventListener('click', () => {

if (confirm('¿Quieres borrar todo el historial acumulado?')) {

series = [];

localStorage.removeItem('gym_logs_pro');

renderSeries();

stopTimer();

}

});


renderSeries();

renderRutinas();
